System Communication 4.1 Event Bus The event bus carries discrete events — things that happened once. Systems subscribe at construction. Events are fired synchronously during simulation ticks. The view layer also subscribes to drive audio and visual effects. public static class EventBus { private static readonly Dictionary&lt;Type, List&lt;Delegate&gt;&gt; _handlers = new(); public static void Subscribe&lt;T&gt;(Action&lt;T&gt; handler) { if (!_handlers.ContainsKey(typeof(T))) _handlers[typeof(T)] = new List&lt;Delegate&gt;(); _handlers[typeof(T)].Add(handler); } public static void Publish&lt;T&gt;(T evt) { if (_handlers.TryGetValue(typeof(T), out var handlers)) foreach (var h in handlers) ((Action&lt;T&gt;)h)(evt); } public static void Unsubscribe&lt;T&gt;(Action&lt;T&gt; handler) { if (_handlers.TryGetValue(typeof(T), out var handlers)) handlers.Remove(handler); } } // Event structs — value types, no allocation public readonly struct BlockPlacedEvent { public readonly Block Block; public readonly Construct Construct; public BlockPlacedEvent(Block b, Construct c) { Block=b; Construct=c; } } public readonly struct BlockDestroyedEvent { public readonly Block Block; public readonly Construct PreviousConstruct; public BlockDestroyedEvent(Block b, Construct c) { Block=b; PreviousConstruct=c; } } public readonly struct ConstructSplitEvent { public readonly Construct Original; public readonly Construct[] NewConstructs; } public readonly struct TierUnlockedEvent { public readonly int Tier; } public readonly struct EnemyKilledEvent { public readonly Enemy Enemy; public readonly Vector3 Position; } public readonly struct PowerShortageEvent { public readonly int NetworkId; public readonly float DeficitKW; } 4.2 Explicit Tick Order Continuous simulation uses shared state and an explicit tick order. Every tick the systems run in the same sequence. This eliminates ordering bugs and makes the simulation deterministic. public class Simulation { // Systems — instantiated once, persist for game lifetime private readonly ConstructSystem _constructs; private readonly PowerSystem _power; private readonly LogisticsSystem _logistics; private readonly FluidSystem _fluids; private readonly MachineSystem _machines; private readonly EnemySystem _enemies; private readonly CombatSystem _combat; private readonly ProgressionSystem _progression; private readonly WorldStreamer _streamer; public void Tick(float dt) { _state.Tick++; // 1. World — load/unload chunks _streamer.Tick(_state, dt); // 2. Construct membership — flood-fill after block changes _constructs.Tick(_state, dt); // 3. Power — balance networks, apply shortage throttle _power.Tick(_state, dt); // 4. Logistics — advance belt slots, run inserters _logistics.Tick(_state, dt); // 5. Fluids — advance fluid networks _fluids.Tick(_state, dt); // 6. Machines — consume inputs, advance production, emit outputs _machines.Tick(_state, dt); // 7. Enemies — AI tick (LOD-gated by distance) _enemies.Tick(_state, dt); // 8. Combat — resolve damage, apply to blocks and entities _combat.Tick(_state, dt); // 9. Progression — check milestones, fire tier unlocks _progression.Tick(_state, dt); } } 5. Key System Implementations 5.1 Construct Membership — Flood Fill When a block is added or removed, the ConstructSystem recalculates membership using a flood fill from the changed block&apos;s neighbours. public class ConstructSystem { public void OnBlockRemoved(Block block, SimulationState state) { var neighbours = GetAdjacentBlocks(block, state.Blocks); if (neighbours.Count == 0) return; // Flood-fill from each neighbour — find connected components var seen = new HashSet&lt;int&gt;(); var components = new List&lt;HashSet&lt;int&gt;&gt;(); foreach (var neighbour in neighbours) { if (seen.Contains(neighbour.Id)) continue; var component = FloodFill(neighbour, state.Blocks, seen); components.Add(component); } if (components.Count == 1) return; // no split — done // Split: first component keeps original construct // Remaining components become new constructs var original = state.Constructs.Get(block.ConstructId); original.BlockIds = components[0]; original.Reclassify(state.Blocks); for (int i = 1; i &lt; components.Count; i++) { var newConstruct = state.Constructs.Create(); newConstruct.BlockIds = components[i]; newConstruct.Reclassify(state.Blocks); foreach (var id in components[i]) state.Blocks.Get(id).ConstructId = newConstruct.Id; } EventBus.Publish(new ConstructSplitEvent { Original = original, NewConstructs = components.Skip(1) .Select(c =&gt; state.Constructs.GetByBlockSet(c)).ToArray() }); } private HashSet&lt;int&gt; FloodFill(Block start, BlockTable blocks, HashSet&lt;int&gt; seen) { var result = new HashSet&lt;int&gt;(); var queue = new Queue&lt;Block&gt;(); queue.Enqueue(start); while (queue.Count &gt; 0) { var current = queue.Dequeue(); if (!seen.Add(current.Id)) continue; result.Add(current.Id); foreach (var n in GetAdjacentBlocks(current, blocks)) if (!seen.Contains(n.Id)) queue.Enqueue(n); } return result; } } 5.2 Power Network Balance public class PowerSystem { public void Tick(SimulationState state, float dt) { foreach (var network in state.PowerNetworks.Active) { float supply = 0f, demand = 0f, turretDemand = 0f; // Sum generators foreach (var id in network.GeneratorIds) { var block = state.Blocks.Get(id); supply += block.GeneratorState.CurrentOutputKW; } // Discharge batteries to cover deficit foreach (var id in network.BatteryIds) supply += DischargeBattery(state.Blocks.Get(id), dt); // Sum consumers — separate turrets foreach (var id in network.ConsumerIds) { var block = state.Blocks.Get(id); if (block.Definition.Category == BlockCategory.Defense) turretDemand += block.Definition.PowerDrawKW; else demand += block.Definition.PowerDrawKW; } float totalDemand = demand + turretDemand; network.CurrentSupply = supply; network.CurrentDemand = totalDemand; // Apply power — turrets protected, others throttled if (supply &gt;= totalDemand) { SetAllOperating(network, state, 1f); ChargeExcessToBatteries(network, state, supply - totalDemand, dt); } else if (supply &gt;= turretDemand) { SetTurretsOperating(network, state, 1f); float ratio = (supply - turretDemand) / demand; SetNonTurretsOperating(network, state, ratio); } else { SetAllOperating(network, state, 0f); EventBus.Publish(new PowerShortageEvent { NetworkId = network.Id, DeficitKW = totalDemand - supply }); } } } } 5.3 Enemy LOD Tick System public class EnemySystem { // Tick intervals by distance from player private const float NEAR_INTERVAL = 0.05f; // 20Hz &lt; 100m private const float MID_INTERVAL = 0.2f; // 5Hz 100-300m private const float FAR_INTERVAL = 1.0f; // 1Hz 300m+ public void Tick(SimulationState state, float dt) { var playerPos = state.Player.Position; foreach (var enemy in state.Enemies.Active) { float dist = Vector3.Distance(enemy.Position, playerPos); float interval = dist &lt; 100f ? NEAR_INTERVAL : dist &lt; 300f ? MID_INTERVAL : FAR_INTERVAL; enemy.TickAccumulator += dt; if (enemy.TickAccumulator &lt; interval) continue; enemy.TickAccumulator -= interval; TickEnemy(enemy, interval, state); } } private void TickEnemy(Enemy enemy, float dt, SimulationState state) { switch (enemy.State) { case EnemyState.Idle: UpdateIdleBehaviour(enemy, dt, state); break; case EnemyState.Alert: UpdateAlert(enemy, dt, state); break; case EnemyState.Attacking: UpdateAttack(enemy, dt, state); break; case EnemyState.Retreating: UpdateRetreat(enemy, dt, state); break; } } } 6. World Streaming The world is a uniform grid of chunks. Each chunk is 32×32 cells = 16m×16m at 0.5m cell size. Two independent radii control simulation and rendering. public class WorldStreamer { private const int CHUNK_SIZE = 32; // cells per side private const int SIM_RADIUS = 5; // chunks — simulation active private const int VIEW_RADIUS = 3; // chunks — GameObjects spawned private const float CELL_SIZE = 0.5f; // metres per cell private Dictionary&lt;Vector2Int, Chunk&gt; _loaded = new(); public void Tick(SimulationState state, float dt) { var centre = WorldToChunk(state.Player.Position); // Expand — load chunks within sim radius for (int x = -SIM_RADIUS; x &lt;= SIM_RADIUS; x++) for (int z = -SIM_RADIUS; z &lt;= SIM_RADIUS; z++) { var coord = centre + new Vector2Int(x, z); if (!_loaded.ContainsKey(coord)) LoadChunk(coord, state); } // Contract — save and evict distant chunks foreach (var coord in _loaded.Keys.ToList()) { if (ChunkDistance(coord, centre) &gt; SIM_RADIUS + 1) { SaveChunk(_loaded[coord]); _loaded.Remove(coord); EventBus.Publish(new ChunkUnloadedEvent { Coord = coord }); } } // Update active list on state state.Chunks.SetActive(_loaded.Values.ToList()); } // Network graphs span chunks — maintained separately // Only nodes in loaded chunks participate in simulation public void OnChunkLoaded(Chunk chunk, SimulationState state) { foreach (var block in chunk.Blocks) _powerNetworkManager.RegisterBlock(block, state); EventBus.Publish(new ChunkLoadedEvent { Chunk = chunk }); } } // View layer responds to chunk events — spawns/despawns GameObjects public class ChunkViewManager : MonoBehaviour { void Start() { EventBus.Subscribe&lt;ChunkLoadedEvent&gt;(OnChunkLoaded); EventBus.Subscribe&lt;ChunkUnloadedEvent&gt;(OnChunkUnloaded); } void OnChunkLoaded(ChunkLoadedEvent e) { if (IsInViewRadius(e.Chunk.Coord)) SpawnChunkView(e.Chunk); } void OnChunkUnloaded(ChunkUnloadedEvent e) { DespawnChunkView(e.Coord); } } 8. Pathfinding 8.1 Flow Fields for Enemies public class FlowField { public Vector2Int Target; public Vector2[,] Vectors; // direction per cell public bool IsDirty; // Built with Dijkstra from target outward — O(n) single pass public void Build(WorldGrid grid) { var open = new Queue&lt;Vector2Int&gt;(); var visited = new bool[grid.Width, grid.Height]; Vectors = new Vector2[grid.Width, grid.Height]; open.Enqueue(Target); visited[Target.x, Target.y] = true; while (open.Count &gt; 0) { var current = open.Dequeue(); foreach (var n in grid.GetNeighbours(current)) { if (visited[n.x, n.y] || grid.IsBlocked(n)) continue; Vectors[n.x, n.y] = ((Vector2)(current - n)).normalized; visited[n.x, n.y] = true; open.Enqueue(n); } } IsDirty = false; } } public class FlowFieldManager { private Dictionary&lt;int, FlowField&gt; _fields = new(); // Each unique target construct gets one field // Shared by ALL enemies targeting that construct public FlowField GetOrCreate(int constructId, Vector2Int target, WorldGrid grid) { if (!_fields.ContainsKey(constructId)) { var f = new FlowField { Target = target }; f.Build(grid); _fields[constructId] = f; } return _fields[constructId]; } // Called when terrain or constructs change in a chunk public void InvalidateChunk(Vector2Int chunkCoord) { foreach (var f in _fields.Values) if (f.CoversChunk(chunkCoord)) f.IsDirty = true; } // Rebuild dirty fields asynchronously public async Task RebuildDirtyFields(WorldGrid grid) { var dirty = _fields.Values.Where(f =&gt; f.IsDirty).ToList(); foreach (var f in dirty) await Task.Run(() =&gt; f.Build(grid)); } } // Enemy movement — just a vector lookup per tick void MoveWithFlowField(Enemy enemy, FlowField field, float dt) { var cell = WorldToCell(enemy.Position); var dir = field.Vectors[cell.x, cell.y]; enemy.Position += new Vector3(dir.x, 0, dir.y) * enemy.Speed * dt; } 9.
